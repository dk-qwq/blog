Q： $\sum_{i = l}^r \lfloor \sqrt i \rfloor$

$1 \leq l, r \leq 10^{12}$

可以发现对于 $sqrt(a) = b$ 相同的一段是连续的，在 $[b^2, (b + 1) ^ 2 - 1]$ 内

那么我们可以预处理出 $sqrt \leq b$ 的数量，这部分预处理复杂度是 $\Omicron(sqrt(|V|))$ 的， 查询复杂度是 $\Omicron(\log sqrt(|V|))$ 的

然后我们确定了 他落在哪个 $sqrt(a) = b$ 的区间上就可以在这个区间内二分；最后差分就可以拿到 $\sum_{i = l}^r \lfloor \sqrt i \rfloor$

```cpp collapse = {1-39}
#include <bits/stdc++.h>
using namespace std;
using ll = long long;
const int N = 1e6 + 5;
ll preSiz[N];
ll preVal[N];
ll floorSqrt(ll x) {
    ll y = sqrt(x);
    while(y * y >= x) y--;
    while((y + 1) * (y + 1) <= x) y++;
    return y;
}
ll solve(ll x) {
    int p = lower_bound(preSiz + 1, preSiz + N, x) - preSiz;
    if(preSiz[p] > x) p--;
    ll c = x - preSiz[p];
    return preVal[p] + c * floorSqrt(x);
}
void solve() {
    ll l, r;
    cin >> l >> r;
    cout << solve(r) - solve(l - 1) << endl;
}
int main() {
    for(int i = 1; i < N; i++) preSiz[i] = 1ll * (i + 1) * (i + 1) - 1ll * i * i;
    for(int i = 1; i < N; i++) preVal[i] = preSiz[i] * i;
       
    for(int i = 1; i < N; i++) preSiz[i] += preSiz[i - 1];
    for(int i = 1; i < N; i++) preVal[i] += preVal[i - 1];


    ios::sync_with_stdio(false);
    cin.tie(nullptr), cout.tie(nullptr);
    // freopen("1.in", "r", stdin);
    // freopen("A.out", "w", stdout);
    int T = 1;
    cin >> T;
    while(T--) solve();
}
```
